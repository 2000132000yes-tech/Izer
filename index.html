const canvas = document.getElementById('gameCanvas'); const ctx = canvas.getContext('2d'); // 공 설정 (기존 반지름 10 -> 4분의 1 크기인 2.5) const ball = { x: canvas.width / 2, y: canvas.height / 2, radius: 2.5, color: 'orange', dx: 4, dy: -4, history: [], maxHistory: 8, active: true }; // 픽셀 격자 설정 const pixelSize = 8; const pixels = []; const cols = Math.floor(canvas.width / pixelSize); const rows = Math.floor(canvas.height / pixelSize); for (let r = 0; r < rows; r++) { for (let c = 0; c < cols; c++) { const px = c * pixelSize; const py = r * pixelSize; // 공의 시작 위치 주변에는 픽셀을 배치하지 않음 const distToBall = Math.hypot((px + pixelSize / 2) - ball.x, (py + pixelSize / 2) - ball.y); if (distToBall > 30) { pixels.push({ x: px, y: py }); } } } // 파티클 (벽 충돌 이펙트) const particles = []; function createImpactParticles(x, y) { for (let i = 0; i < 20; i++) { const angle = Math.random() * Math.PI * 2; const speed = Math.random() * 4 + 1; particles.push({ x: x, y: y, dx: Math.cos(angle) * speed, dy: Math.sin(angle) * speed, radius: Math.random() * 2 + 1, color: 'orange', alpha: 1, life: 0.05 }); } } function update() { if (!ball.active) { // 파티클 업데이트 for (let i = particles.length - 1; i >= 0; i--) { const p = particles[i]; p.x += p.dx; p.y += p.dy; p.alpha -= p.life; if (p.alpha <= 0) { particles.splice(i, 1); } } return; } // 잔상 기록 ball.history.push({ x: ball.x, y: ball.y }); if (ball.history.length > ball.maxHistory) { ball.history.shift(); } // 공 이동 ball.x += ball.dx; ball.y += ball.dy; // 벽 충돌 감지 및 제거 이펙트 if ( ball.x - ball.radius <= 0 || ball.x + ball.radius >= canvas.width || ball.y - ball.radius <= 0 || ball.y + ball.radius >= canvas.height ) { const impactX = Math.max(0, Math.min(canvas.width, ball.x)); const impactY = Math.max(0, Math.min(canvas.height, ball.y)); createImpactParticles(impactX, impactY); ball.active = false; } } function draw() { // 배경 (검은색 + 잔상 표현을 위한 알파 블렌딩) ctx.fillStyle = 'rgba(0, 0, 0, 0.3)'; ctx.fillRect(0, 0, canvas.width, canvas.height); // 픽셀 그리기 ctx.fillStyle = '#333333'; for (let i = 0; i < pixels.length; i++) { ctx.fillRect(pixels[i].x, pixels[i].y, pixelSize - 1, pixelSize - 1); } // 공 및 잔상 그리기 if (ball.active) { for (let i = 0; i < ball.history.length; i++) { const pos = ball.history[i]; const alpha = (i + 1) / ball.history.length; ctx.beginPath(); ctx.arc(pos.x, pos.y, ball.radius, 0, Math.PI * 2); ctx.fillStyle = `rgba(255, 165, 0, ${alpha * 0.5})`; ctx.fill(); ctx.closePath(); } ctx.beginPath(); ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2); ctx.fillStyle = ball.color; ctx.fill(); ctx.closePath(); } // 파티클 그리기 for (let i = 0; i < particles.length; i++) { const p = particles[i]; ctx.beginPath(); ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2); ctx.fillStyle = `rgba(255, 165, 0, ${p.alpha})`; ctx.fill(); ctx.closePath(); } } function loop() { update();
